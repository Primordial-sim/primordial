extends Camera2D

var _world_width: int = 100
var _world_height: int = 100
var _cell_size: float = 8.0
var _is_panning: bool = false
var _pan_start: Vector2 = Vector2.ZERO
var _camera_start: Vector2 = Vector2.ZERO

const MIN_ZOOM: float = 1.0
const MAX_ZOOM: float = 4.0
const ZOOM_STEP: float = 0.25
const VIEWPORT_SIZE: float = 800.0

func set_world_dimensions(w: int, h: int) -> void:
    _world_width = w
    _world_height = h

func set_cell_size(s: float) -> void:
    _cell_size = s

func _get_world_size() -> Vector2:
    return Vector2(float(_world_width) * _cell_size, float(_world_height) * _cell_size)

# Función pública para que otros scripts puedan forzar el wrapping
func wrap_camera_position() -> void:
    var world_w = float(_world_width) * _cell_size
    var world_h = float(_world_height) * _cell_size
    var center_x = world_w / 2.0
    var center_y = world_h / 2.0
    
    var offset_x = position.x - center_x
    var offset_y = position.y - center_y
    
    offset_x = fmod(offset_x, world_w)
    offset_y = fmod(offset_y, world_h)
    
    if offset_x > world_w / 2.0:
        offset_x -= world_w
    elif offset_x < -world_w / 2.0:
        offset_x += world_w
    
    if offset_y > world_h / 2.0:
        offset_y -= world_h
    elif offset_y < -world_h / 2.0:
        offset_y += world_h
    
    position.x = center_x + offset_x
    position.y = center_y + offset_y

func reset_view() -> void:
    position = _get_world_size() / 2.0
    zoom = Vector2.ONE
    wrap_camera_position()

func _ready() -> void:
    make_current()
    reset_view()

func _unhandled_input(event: InputEvent) -> void:
    if event is InputEventMouseButton:
        var mouse_event: InputEventMouseButton = event
        if mouse_event.button_index == MOUSE_BUTTON_WHEEL_UP:
            _zoom_at_position(zoom.x + ZOOM_STEP, mouse_event.position)
        elif mouse_event.button_index == MOUSE_BUTTON_WHEEL_DOWN:
            _zoom_at_position(zoom.x - ZOOM_STEP, mouse_event.position)
        elif mouse_event.button_index == MOUSE_BUTTON_RIGHT:
            _is_panning = mouse_event.pressed
            if _is_panning:
                _pan_start = mouse_event.position
                _camera_start = position
    elif event is InputEventMouseMotion and _is_panning:
        var motion: InputEventMouseMotion = event
        position = _camera_start - (motion.position - _pan_start) / zoom
        wrap_camera_position()
    elif event is InputEventKey:
        var key_event: InputEventKey = event
        if key_event.pressed and key_event.keycode == KEY_C:
            reset_view()

func _zoom_at_position(new_zoom: float, screen_pos: Vector2) -> void:
    var old_zoom = zoom.x
    new_zoom = clamp(new_zoom, MIN_ZOOM, MAX_ZOOM)
    if abs(new_zoom - old_zoom) < 0.01: return
    
    var world_pos_under_cursor = (screen_pos - Vector2(VIEWPORT_SIZE / 2.0, VIEWPORT_SIZE / 2.0)) / old_zoom + position
    zoom = Vector2(new_zoom, new_zoom)
    position = world_pos_under_cursor - (screen_pos - Vector2(VIEWPORT_SIZE / 2.0, VIEWPORT_SIZE / 2.0)) / new_zoom
    wrap_camera_position()

func _draw_group_inspector(viewport_size: Vector2) -> void:
    if _group_agents.is_empty(): return
    var panel_x: float = viewport_size.x - INSPECTOR_WIDTH - 20.0
    var panel_y: float = 60.0
    var panel_height: float = 10.0 * INSPECTOR_LINE_HEIGHT + INSPECTOR_PADDING * 2.0 + 30.0
    var panel_rect: Rect2 = Rect2(Vector2(panel_x, panel_y), Vector2(INSPECTOR_WIDTH, panel_height))
    
    draw_rect(panel_rect, INSPECTOR_BG_COLOR)
    draw_rect(panel_rect, INSPECTOR_BORDER_COLOR, false, 1.0)
    
    var title_y: float = panel_y + INSPECTOR_PADDING + 14.0
    draw_string(ThemeDB.fallback_font, Vector2(panel_x + INSPECTOR_PADDING, title_y), "👥 GRUPO (%d)" % _group_agents.size(), HORIZONTAL_ALIGNMENT_LEFT, -1, 13, INSPECTOR_TITLE_COLOR)
    
    var sep_y: float = title_y + 8.0
    draw_line(Vector2(panel_x + INSPECTOR_PADDING, sep_y), Vector2(panel_x + INSPECTOR_WIDTH - INSPECTOR_PADDING, sep_y), INSPECTOR_BORDER_COLOR, 1.0)
    
    var total_age: float = 0.0
    var males: int = 0; var females: int = 0; var sick_count: int = 0
    var adults: int = 0; var seniors: int = 0
    var species_count: Dictionary = {}
    
    for agent in _group_agents:
        total_age += agent.get("age", 0.0)
        if agent.get("gender", "") == "M": males += 1
        elif agent.get("gender", "") == "F": females += 1
        if agent.get("is_sick", false): sick_count += 1
        if agent.get("is_adult", true): adults += 1
        if agent.get("is_senior", false): seniors += 1

                    var drag_distance = world_pos.distance_to(_drag_start_pos)
                    
                    if drag_distance < 5.0:

                        if not clicked_agent.is_empty():

            var status: String = data.get("status", "")

            elif status == "running": _is_paused = false
            elif status.begins_with("speed:"): _current_speed = float(status.substr(6))
            ui_renderer.update_control_state(_is_paused, _current_speed)

    # Calcular el rectángulo de selección
    var top_left = Vector2(min(_selection_rect_start.x, _selection_rect_end.x), 
                           min(_selection_rect_start.y, _selection_rect_end.y))
    var bottom_right = Vector2(max(_selection_rect_start.x, _selection_rect_end.x),

        var ax = agent.get("x", 0.0) * _cell_size

        # Si la celda del agente intersecta con el rectángulo de selección, se selecciona
        if selection_rect.intersects(agent_rect) or selection_rect.has_point(agent_rect.get_center()):
